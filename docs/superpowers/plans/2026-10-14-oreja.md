# Oreja Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ein Windows-Tray-Tool, das per Klick Mikrofon und Systemaudio (alle Wiedergabegeräte) gemischt als FLAC (16 kHz, Mono, 16 Bit) aufnimmt und fertige Dateien in einen wählbaren Ordner (z. B. Google Drive) verschiebt.

**Architecture:** WASAPI-Quellen (NAudio) wandeln im Capture-Thread nach 16 kHz Mono Float und schreiben in je einen Jitter-Puffer (`SampleFifo`). Ein Taktgeber-Thread (`AudioMixer`, alle 50 ms nach Echtzeit) mischt alle Puffer, füllt Lücken mit Stille und gibt 16-Bit-PCM an `LibFlacSink` (libFLAC per P/Invoke). Der `Recorder` steuert eine Aufnahme und folgt Gerätewechseln, die `TrayApp` (WinForms `NotifyIcon`) zeigt nur Zustände an, der `RecordingMover` verschiebt fertige Dateien aus dem lokalen Arbeitsordner in den Zielordner.

**Tech Stack:** C# / .NET 10 (`net10.0-windows`, WinForms), NAudio 3.1.0 (`NAudio.Wasapi`), libFLAC 1.5.0 (offizielles Win64-Build von Xiph), xUnit 2.9.3.

**Spec:** `docs/superpowers/specs/2026-10-14-oreja-design.md`

## Global Constraints

- Zielplattform `net10.0-windows`, `win-x64`, Windows 10 22H2 und Windows 11; Auslieferung als self-contained Single-File-Exe (komprimiert, native DLL eingebettet).
- Pakete: `NAudio.Wasapi` 3.1.0 (bringt `NAudio.Core` mit). Keine weiteren Laufzeitabhängigkeiten außer `libFLAC.dll` 1.5.0 (Win64 aus `flac-1.5.0-win.zip`, SHA-256 des Zips `53f1500f0d6e7c61379d7fee50d4a9f7f504c650009506d9ba015530d76c0dde`, der DLL `f93499172875fc2c0df80b57086f32e3f39e835283952ee2a59a3d4ffb097644`).
- Audioformat der Datei: FLAC, 16 000 Hz, 1 Kanal, 16 Bit.
- Quellen: Windows-Standardaufnahmegerät (`DataFlow.Capture`, `Role.Console`) plus Loopback **jedes** aktiven Wiedergabegeräts, gemischt in eine Spur.
- Taktgeber alle 50 ms; Vorlauf je Quelle 200 ms = 3200 Samples; Höchstfüllstand 1 s = 16 000 Samples, darüber verwerfen bis auf 3200.
- Mischen: Summe, begrenzt auf [-1, 1], dann `short` per `Math.Round(x * 32767)`.

```csharp
        }
```

```csharp
        File.WriteAllText(path, "x");

        Assert.Throws<IOException>(() => LibFlacSink.Create(path));
    }

    [Fact]
    public void WriteFailureSurfacesAsIOException()
    {
        var sink = new LibFlacSink(new FailingStream(failAfterBytes: 4000));
        var chunk = TestSignal(800);

        var error = Assert.Throws<IOException>(() =>
        {
            for (int i = 0; i < 100; i++) sink.Write(chunk);
        });

        Assert.Equal("Datenträger voll", error.InnerException?.Message);
        sink.Dispose(); // darf nach einem Schreibfehler nicht werfen
    }

    private static short[] TestSignal(int count)
    {
        var samples = new short[count];
        for (int i = 0; i < count; i++)
            samples[i] = (short)(Math.Sin(2 * Math.PI * 440 * i / 16000.0) * 12000 + (i % 7) - 3);
        return samples;

        Assert.Equal(16000, reader.WaveFormat.SampleRate);
        Assert.Equal(1, reader.WaveFormat.Channels);
        Assert.Equal(16, reader.WaveFormat.BitsPerSample);

        reader.CopyTo(pcm);
        var bytes = pcm.ToArray();
        var samples = new short[bytes.Length / 2];
        Buffer.BlockCopy(bytes, 0, samples, 0, samples.Length * 2);
        return samples;
    }

        public override void Write(ReadOnlySpan<byte> buffer)
        {
            if (Length + buffer.Length > failAfterBytes) throw new IOException("Datenträger voll");
            base.Write(buffer);
        }
    }
```

`src/Oreja/Recording/LibFlacSink.cs`:

```csharp
using System.Runtime.CompilerServices;
using System.Runtime.InteropServices;
using static Oreja.Recording.LibFlacNative;

namespace Oreja.Recording;

/// <summary>
/// Kodiert 16-Bit-Mono-PCM mit libFLAC. Jeder fertige FLAC-Frame wird sofort in den Stream
/// geschrieben und geflusht, damit ein Absturz höchstens die letzten Sekunden kostet.
/// </summary>
public sealed unsafe class LibFlacSink : IFlacSink
{
    private const int CallbackOk = 0;
    private const int CallbackError = 1;

    private readonly Stream _stream;
    private GCHandle _self;
    private IntPtr _encoder;
    private int[] _buffer = new int[4096];
    private Exception? _ioError;
    private bool _completed;

    public static LibFlacSink Create(string path, int sampleRate = 16000) =>
        new(new FileStream(path, FileMode.CreateNew, FileAccess.ReadWrite, FileShare.Read), sampleRate);

    public LibFlacSink(Stream stream, int sampleRate = 16000)
    {
        _stream = stream;
        _encoder = FLAC__stream_encoder_new();
        if (_encoder == IntPtr.Zero)
        {
            stream.Dispose();
            throw new InvalidOperationException("libFLAC: Encoder konnte nicht angelegt werden");
        }

        _self = GCHandle.Alloc(this);
        FLAC__stream_encoder_set_channels(_encoder, 1);
        FLAC__stream_encoder_set_bits_per_sample(_encoder, 16);
        FLAC__stream_encoder_set_sample_rate(_encoder, (uint)sampleRate);
        FLAC__stream_encoder_set_compression_level(_encoder, 5);

        int status = FLAC__stream_encoder_init_stream(_encoder, &OnWrite, &OnSeek, &OnTell, IntPtr.Zero, GCHandle.ToIntPtr(_self));
        if (status != 0)
        {
            var error = _ioError;
            Dispose();
            throw new IOException($"libFLAC: Initialisierung fehlgeschlagen (Status {status})", error);
        }
    }

    public void Write(ReadOnlySpan<short> samples)
    {
        ObjectDisposedException.ThrowIf(_encoder == IntPtr.Zero, this);
        if (samples.IsEmpty) return;
        if (_buffer.Length < samples.Length) _buffer = new int[samples.Length];
        for (int i = 0; i < samples.Length; i++) _buffer[i] = samples[i];

        fixed (int* p = _buffer)
        {
            if (FLAC__stream_encoder_process_interleaved(_encoder, p, (uint)samples.Length) == 0)
                throw new IOException("libFLAC: Kodieren fehlgeschlagen", _ioError);
        }
    }

    public void Complete()
    {
        ObjectDisposedException.ThrowIf(_encoder == IntPtr.Zero, this);
        if (_completed) return;
        _completed = true;
        if (FLAC__stream_encoder_finish(_encoder) == 0)
            throw new IOException("libFLAC: Abschließen der Datei fehlgeschlagen", _ioError);
        _stream.Flush();
    }

    public void Dispose()
    {
        if (_encoder != IntPtr.Zero)
        {
            // delete() schließt den Stream ab, falls Complete() nicht aufgerufen wurde.
            FLAC__stream_encoder_delete(_encoder);
            _encoder = IntPtr.Zero;
        }
        if (_self.IsAllocated) _self.Free();
        _stream.Dispose();
    }

    private static LibFlacSink From(IntPtr clientData) => (LibFlacSink)GCHandle.FromIntPtr(clientData).Target!;

    // Die Callbacks dürfen nie werfen: Eine Ausnahme, die in nativen Code läuft, beendet den Prozess.
    [UnmanagedCallersOnly(CallConvs = [typeof(CallConvCdecl)])]
    private static int OnWrite(IntPtr encoder, byte* buffer, nuint bytes, uint samples, uint currentFrame, IntPtr clientData)
    {
        var sink = From(clientData);
        try
        {
            sink._stream.Write(new ReadOnlySpan<byte>(buffer, checked((int)bytes)));
            sink._stream.Flush();
            return CallbackOk;
        }
        catch (Exception ex)
        {
            sink._ioError = ex;
            return CallbackError;
        }
    }

    [UnmanagedCallersOnly(CallConvs = [typeof(CallConvCdecl)])]
    private static int OnSeek(IntPtr encoder, ulong offset, IntPtr clientData)
    {
        var sink = From(clientData);
        try
        {
            sink._stream.Position = checked((long)offset);
            return CallbackOk;
        }
        catch (Exception ex)
        {
            sink._ioError = ex;
            return CallbackError;
        }
    }

    [UnmanagedCallersOnly(CallConvs = [typeof(CallConvCdecl)])]
    private static int OnTell(IntPtr encoder, ulong* offset, IntPtr clientData)
    {
        var sink = From(clientData);
        try
        {
            *offset = checked((ulong)sink._stream.Position);
            return CallbackOk;
        }
        catch (Exception ex)
        {
            sink._ioError = ex;
            return CallbackError;
        }
    }
}
```

```csharp
namespace Oreja.Recording;

/// <summary>
/// Jitter-Puffer einer Quelle im Zielformat (16 kHz Mono Float). Wird vom Capture-Thread
/// beschrieben und vom Mixer-Thread gelesen.
/// Zustände: „Füllen“ liefert Stille, bis der Vorlauf erreicht ist; „Abspielen“ liest,
/// bis der Puffer leerläuft. Über dem Höchstfüllstand wird das Älteste verworfen.
/// </summary>
public sealed class SampleFifo
{
    private readonly object _gate = new();
    private readonly float[] _buffer;
    private readonly int _prebuffer;
    private readonly int _max;
    private int _head;
    private int _count;
    private bool _playing;
    private int _overflows;
    private int _underruns;

    public SampleFifo(int prebufferSamples, int maxSamples)
    {
        ArgumentOutOfRangeException.ThrowIfNegativeOrZero(prebufferSamples);
        ArgumentOutOfRangeException.ThrowIfLessThan(maxSamples, prebufferSamples);
        _prebuffer = prebufferSamples;
        _max = maxSamples;
        _buffer = new float[maxSamples * 2];
    }

    public int Count { get { lock (_gate) return _count; } }
    public bool IsPlaying { get { lock (_gate) return _playing; } }
    public int OverflowCount { get { lock (_gate) return _overflows; } }
    public int UnderrunCount { get { lock (_gate) return _underruns; } }

    public void Write(ReadOnlySpan<float> samples)
    {
        lock (_gate)
        {
            if (samples.Length > _buffer.Length) samples = samples[^_buffer.Length..];
            int free = _buffer.Length - _count;
            if (samples.Length > free) Discard(samples.Length - free);

            int tail = (_head + _count) % _buffer.Length;
            int first = Math.Min(samples.Length, _buffer.Length - tail);
            samples[..first].CopyTo(_buffer.AsSpan(tail));
            samples[first..].CopyTo(_buffer);
            _count += samples.Length;

            if (_count > _max)
            {
                Discard(_count - _prebuffer);
                _overflows++;
            }
        }
    }

    /// <summary>Addiert bis zu destination.Length Samples auf destination. Im Vorlauf passiert nichts (Stille).</summary>
    public void MixInto(Span<float> destination)
    {
        lock (_gate)
        {
            if (!_playing)
            {
                if (_count < _prebuffer) return;
                _playing = true;
            }

            int n = Math.Min(destination.Length, _count);
            AddFromBuffer(destination[..n]);
            if (n < destination.Length)
            {
                _playing = false;
                _underruns++;
            }
        }
    }

    /// <summary>Addiert alles Gepufferte (auch im Vorlauf) auf destination; liefert die Anzahl Samples.</summary>
    public int DrainInto(Span<float> destination)
    {
        lock (_gate)
        {
            int n = Math.Min(destination.Length, _count);
            AddFromBuffer(destination[..n]);
            _playing = false;
            return n;
        }
    }

    private void AddFromBuffer(Span<float> destination)
    {
        for (int i = 0; i < destination.Length; i++)
            destination[i] += _buffer[(_head + i) % _buffer.Length];
        Discard(destination.Length);
    }

    private void Discard(int n)
    {
        _head = (_head + n) % _buffer.Length;
        _count -= n;
    }
}
```

```csharp
using Oreja.Infrastructure;

namespace Oreja.Recording;

public delegate void PcmWriter(ReadOnlySpan<short> samples);

/// <summary>
/// Taktgeber und Mischer: holt bei jedem Tick so viele Samples aus allen Quellen-FIFOs,
/// wie laut Uhr seit Start fällig sind, addiert sie, begrenzt auf [-1, 1] und gibt
/// 16-Bit-PCM aus. Quellen dürfen während der Aufnahme hinzukommen oder wegfallen.
/// </summary>
public sealed class AudioMixer
{
    public const int SampleRate = 16000;
    public const int PrebufferSamples = SampleRate / 5; // 200 ms
    public const int MaxBufferedSamples = SampleRate;   // 1 s

    private readonly IClock _clock;
    private readonly PcmWriter _output;
    private readonly object _gate = new();
    private readonly List<SampleFifo> _sources = [];
    private readonly float[] _mix = new float[SampleRate];
    private short[] _pcm = new short[SampleRate];
    private TimeSpan _start;
    private long _written;
    private bool _started;
    private int _removedOverflows;

    public AudioMixer(IClock clock, PcmWriter output)
    {
        _clock = clock;
        _output = output;
    }

    public long SamplesWritten { get { lock (_gate) return _written; } }

    public int OverflowCount
    {
        get { lock (_gate) return _removedOverflows + _sources.Sum(s => s.OverflowCount); }
    }

    public SampleFifo AddSource()
    {
        var fifo = new SampleFifo(PrebufferSamples, MaxBufferedSamples);
        lock (_gate) _sources.Add(fifo);
        return fifo;
    }

    public void RemoveSource(SampleFifo fifo)
    {
        lock (_gate)
        {
            if (_sources.Remove(fifo)) _removedOverflows += fifo.OverflowCount;
        }
    }

    public void Start()
    {
        lock (_gate)
        {
            _start = _clock.Elapsed;
            _written = 0;
            _started = true;
        }
    }

    /// <summary>Gibt alle laut Uhr fälligen Samples aus (in Blöcken von höchstens 1 s).</summary>
    public void Tick()
    {
        lock (_gate)
        {
            if (!_started) throw new InvalidOperationException("Start() wurde nicht aufgerufen");
            long due = (_clock.Elapsed - _start).Ticks * SampleRate / TimeSpan.TicksPerSecond;
            long pending = due - _written;
            while (pending > 0)
            {
                var mix = _mix.AsSpan(0, (int)Math.Min(pending, _mix.Length));
                mix.Clear();
                foreach (var source in _sources) source.MixInto(mix);
                Emit(mix);
                pending -= mix.Length;
            }
        }
    }

    /// <summary>Beim Stoppen: fällige Samples plus alles, was noch in den Puffern liegt.</summary>
    public void Flush()
    {
        lock (_gate)
        {
            Tick();
            int remaining = _sources.Count == 0 ? 0 : _sources.Max(s => s.Count);
            if (remaining == 0) return;
            var mix = (remaining <= _mix.Length ? _mix : new float[remaining]).AsSpan(0, remaining);
            mix.Clear();
            foreach (var source in _sources) source.DrainInto(mix);
            Emit(mix);
        }
    }

    private void Emit(Span<float> mix)
    {
        if (_pcm.Length < mix.Length) _pcm = new short[mix.Length];
        var pcm = _pcm.AsSpan(0, mix.Length);
        for (int i = 0; i < mix.Length; i++)
            pcm[i] = (short)Math.Round(Math.Clamp(mix[i], -1f, 1f) * short.MaxValue);
        _output(pcm);
        _written += mix.Length;
    }
}
```

```csharp
using System.Buffers.Binary;
using NAudio.Dsp;
using NAudio.Wave;

namespace Oreja.Recording;

/// <summary>Wandelt Capture-Puffer im Geräteformat in 16 kHz Mono Float (Kanäle gemittelt, WDL-Resampler).</summary>
public sealed class SampleConverter
{
    private static readonly Guid PcmSubFormat = new("00000001-0000-0010-8000-00aa00389b71");
    private static readonly Guid FloatSubFormat = new("00000003-0000-0010-8000-00aa00389b71");

    private enum SampleKind { Float32, Pcm16, Pcm24, Pcm32 }

    private readonly SampleKind _kind;
    private readonly int _channels;
    private readonly int _blockAlign;
    private readonly int _bytesPerSample;
    private readonly int _inputRate;
    private readonly int _outputRate;
    private readonly WdlResampler? _resampler;
    private float[] _mono = new float[4096];
    private float[] _output = new float[4096];

    public SampleConverter(WaveFormat input, int outputRate = AudioMixer.SampleRate)
    {
        _kind = Classify(input);
        _channels = input.Channels;
        _blockAlign = input.BlockAlign;
        _bytesPerSample = input.BlockAlign / input.Channels;
        _inputRate = input.SampleRate;
        _outputRate = outputRate;

        if (_inputRate != _outputRate)
        {
            _resampler = new WdlResampler();
            _resampler.SetMode(true, 2, false);
            _resampler.SetFilterParms();
            _resampler.SetFeedMode(true);
            _resampler.SetRates(_inputRate, _outputRate);
        }
    }

    /// <summary>Liefert die gewandelten Samples; die Span bleibt bis zum nächsten Aufruf gültig.</summary>
    public ReadOnlySpan<float> Convert(ReadOnlySpan<byte> input)
    {
        int frames = input.Length / _blockAlign;
        if (frames == 0) return ReadOnlySpan<float>.Empty;
        if (_mono.Length < frames) _mono = new float[frames];

        for (int f = 0; f < frames; f++)
        {
            float sum = 0;
            for (int c = 0; c < _channels; c++)
                sum += ReadSample(input.Slice(f * _blockAlign + c * _bytesPerSample, _bytesPerSample));
            _mono[f] = sum / _channels;
        }

        if (_resampler is null) return _mono.AsSpan(0, frames);

        int needed = _resampler.ResamplePrepare(frames, 1, out Span<float> inBuffer);
        _mono.AsSpan(0, needed).CopyTo(inBuffer);
        int capacity = (int)((long)frames * _outputRate / _inputRate) + 16;
        if (_output.Length < capacity) _output = new float[capacity];
        int produced = _resampler.ResampleOut(_output, needed, _output.Length, 1);
        return _output.AsSpan(0, produced);
    }

    private float ReadSample(ReadOnlySpan<byte> b) => _kind switch
    {
        SampleKind.Float32 => BinaryPrimitives.ReadSingleLittleEndian(b),
        SampleKind.Pcm16 => BinaryPrimitives.ReadInt16LittleEndian(b) / 32768f,
        SampleKind.Pcm24 => ((b[2] << 24) | (b[1] << 16) | (b[0] << 8)) / 2147483648f,
        SampleKind.Pcm32 => BinaryPrimitives.ReadInt32LittleEndian(b) / 2147483648f,
        _ => 0f,
    };

    private static SampleKind Classify(WaveFormat format)
    {
        bool isFloat = format.Encoding == WaveFormatEncoding.IeeeFloat
            || (format is WaveFormatExtensible ext && ext.SubFormat == FloatSubFormat);
        bool isPcm = format.Encoding == WaveFormatEncoding.Pcm
            || (format is WaveFormatExtensible pcm && pcm.SubFormat == PcmSubFormat);

        return (isFloat, isPcm, format.BitsPerSample) switch
        {
            (true, _, 32) => SampleKind.Float32,
            (_, true, 16) => SampleKind.Pcm16,
            (_, true, 24) => SampleKind.Pcm24,
            (_, true, 32) => SampleKind.Pcm32,
            _ => throw new NotSupportedException($"Nicht unterstütztes Aufnahmeformat: {format}"),
        };
    }
}
```

```csharp
using Oreja.Infrastructure;

namespace Oreja.Recording;

/// <summary>
/// Erkennt ein Mikrofon ohne Signal: nur exakte digitale Nullen oder gar keine Daten
/// (stummgeschaltetes Headset, in den Windows-Datenschutzeinstellungen gesperrt).
/// Ein echtes Mikrofon rauscht immer minimal.
/// </summary>
public sealed class MicSilenceDetector
{
    public static readonly TimeSpan DefaultThreshold = TimeSpan.FromSeconds(3);

    private readonly IClock _clock;
    private readonly TimeSpan _threshold;
    private long _lastSignalTicks;

    public MicSilenceDetector(IClock clock, TimeSpan threshold)
    {
        _clock = clock;
        _threshold = threshold;
        Reset();
    }

    public void Reset() => Interlocked.Exchange(ref _lastSignalTicks, _clock.Elapsed.Ticks);

    public void Observe(ReadOnlySpan<float> samples)
    {
        foreach (var sample in samples)
        {
            if (sample != 0f)
            {
                Interlocked.Exchange(ref _lastSignalTicks, _clock.Elapsed.Ticks);
                return;
            }
        }
    }

    public bool IsSilent => _clock.Elapsed.Ticks - Interlocked.Read(ref _lastSignalTicks) >= _threshold.Ticks;
}
```

```csharp
using System.Globalization;

namespace Oreja.Infrastructure;

/// <summary>Threadsicheres Zeilenprotokoll mit einer Rotationsstufe (oreja.log → oreja.log.1).</summary>
public sealed class Log
{
    private readonly object _gate = new();
    private readonly string _path;
    private readonly long _maxBytes;

    public Log(string path, long maxBytes = 1_000_000)
    {
        _path = path;
        _maxBytes = maxBytes;
        Directory.CreateDirectory(Path.GetDirectoryName(path)!);
    }

    public void Info(string message) => Write("INFO", message);

    public void Error(string message, Exception? ex = null) =>
        Write("ERROR", ex is null ? message : $"{message}: {ex}");

    private void Write(string level, string message)
    {
        lock (_gate)
        {
            try
            {
                var file = new FileInfo(_path);
                if (file.Exists && file.Length >= _maxBytes) File.Move(_path, _path + ".1", overwrite: true);
                string timestamp = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss.fff", CultureInfo.InvariantCulture);
                File.AppendAllText(_path, $"{timestamp} {level} {message}{Environment.NewLine}");
            }
            catch (Exception ex) when (ex is IOException or UnauthorizedAccessException)
            {
                // Protokollieren darf die App nie zum Absturz bringen.
            }
        }
    }
}
```

```csharp
/// <summary>Wo Aufnahmen entstehen (Arbeitsordner) und wo sie landen (Zielordner).</summary>
public sealed class OutputPaths
{
    public const string Extension = ".flac";

    private readonly Func<string> _targetFolder;

    public OutputPaths(string workingFolder, Func<string> targetFolder)
    {
        WorkingFolder = workingFolder;
        _targetFolder = targetFolder;
    }

    public string WorkingFolder { get; }

    public string TargetFolder => _targetFolder();

    /// <summary>Neue Datei im Arbeitsordner, benannt nach der Startzeit, eindeutig in Arbeits- und Zielordner.</summary>
    public string NewRecordingPath(DateTime start)
    {
        Directory.CreateDirectory(WorkingFolder);
        string target = TargetFolder;
        string baseName = start.ToString("yyyy-MM-dd_HH-mm-ss", CultureInfo.InvariantCulture);
        string name = UniqueFileName(baseName,
            n => File.Exists(Path.Combine(WorkingFolder, n)) || File.Exists(Path.Combine(target, n)));
        return Path.Combine(WorkingFolder, name);
    }

    /// <summary>„name.flac“, bei Kollision „name_2.flac“, „name_3.flac“, …</summary>
    public static string UniqueFileName(string baseName, Func<string, bool> isTaken)
    {
        string name = baseName + Extension;
        for (int i = 2; isTaken(name); i++) name = $"{baseName}_{i}{Extension}";
        return name;
    }
}
```

```csharp
namespace Oreja.Infrastructure;

/// <summary>
/// Verschiebt fertige Aufnahmen vom Arbeitsordner in den Zielordner: Kopie als „name.flac.tmp“
/// ins Ziel, dort umbenennen, erst dann lokal löschen. Sync-Clients (Google Drive) sehen so
/// nur vollständige Dateien. Dateien, die noch geschrieben werden, bleiben liegen.
/// </summary>
public sealed class RecordingMover
{
    private readonly OutputPaths _paths;
    private readonly Log _log;
    private readonly object _gate = new();

    public RecordingMover(OutputPaths paths, Log log)
    {
        _paths = paths;
        _log = log;
    }

    /// <summary>Liefert die Anzahl verschobener Dateien. I/O-Probleme werden protokolliert und beim nächsten Mal erneut versucht.</summary>
    public int MovePending()
    {
        lock (_gate)
        {
            string working = _paths.WorkingFolder;
            string target = _paths.TargetFolder;
            if (!Directory.Exists(working) || SameFolder(working, target)) return 0;

            int moved = 0;
            foreach (var source in Directory.EnumerateFiles(working, "*" + OutputPaths.Extension).Order().ToList())
            {
                try
                {
                    if (MoveOne(source, target)) moved++;
                }
                catch (Exception ex) when (ex is IOException or UnauthorizedAccessException)
                {
                    _log.Error($"Verschieben von {source} nach {target} fehlgeschlagen, neuer Versuch später", ex);
                }
            }
            return moved;
        }
    }

    private bool MoveOne(string source, string targetFolder)
    {
        string fileName = Path.GetFileName(source);
        string temporary = Path.Combine(targetFolder, fileName + ".tmp");

        using (var input = TryOpenExclusive(source))
        {
            if (input is null) return false; // wird noch aufgenommen
            Directory.CreateDirectory(targetFolder);
            using var output = new FileStream(temporary, FileMode.Create, FileAccess.Write, FileShare.None);
            input.CopyTo(output);
        }

        string finalName = OutputPaths.UniqueFileName(Path.GetFileNameWithoutExtension(fileName),
            n => File.Exists(Path.Combine(targetFolder, n)));
        string destination = Path.Combine(targetFolder, finalName);
        File.Move(temporary, destination);
        File.Delete(source);
        _log.Info($"Aufnahme verschoben: {source} -> {destination}");
        return true;
    }

    private static FileStream? TryOpenExclusive(string path)
    {
        try
        {
            return new FileStream(path, FileMode.Open, FileAccess.Read, FileShare.None);
        }
        catch (IOException)
        {
            return null; // noch geöffnet (Freigabeverletzung) oder inzwischen verschwunden
        }
    }

    private static bool SameFolder(string a, string b) =>
        string.Equals(Path.TrimEndingDirectorySeparator(Path.GetFullPath(a)),
                      Path.TrimEndingDirectorySeparator(Path.GetFullPath(b)),
                      StringComparison.OrdinalIgnoreCase);
}
```

```csharp
    private readonly bool _runPumpThread;
    private readonly object _gate = new();
    private RecorderStatus _status = RecorderStatus.Idle;
    private Session? _session;

    public Recorder(IAudioBackend backend, IClock clock, Func<string, IFlacSink> createSink, OutputPaths paths, Log log,
        Action<Action>? dispatch = null, bool runPumpThread = true)
    {
        _backend = backend;
        _clock = clock;
        _createSink = createSink;
        _paths = paths;
        _log = log;
        _dispatch = dispatch ?? DispatchToThreadPool;
        _runPumpThread = runPumpThread;

        // Benachrichtigungen kommen auf fremden Threads; die Arbeit passiert asynchron, damit
        // nie ein Capture- oder COM-Thread auf das Lock wartet.
        _backend.RenderDeviceActivated += id => _dispatch(() => OnRenderDeviceActivated(id));
        _backend.DeviceDeactivated += id => _dispatch(() => OnDeviceDeactivated(id));
        _backend.DefaultCaptureDeviceChanged += id => _dispatch(() => OnDefaultCaptureDeviceChanged(id));
    }

    public RecorderStatus Status { get { lock (_gate) return _status; } }

    /// <summary>Bei jeder Zustandsänderung (beliebiger Thread).</summary>
    public event Action<RecorderStatus>? StatusChanged;

    /// <summary>Wenn eine Aufnahmedatei abgeschlossen ist (Pfad im Arbeitsordner).</summary>
    public event Action<string>? RecordingFinished;

    public void Start()
    {
        RecorderStatus status;
        lock (_gate)
        {
            if (_session is not null) return;
            status = _status = StartLocked();
        }
        StatusChanged?.Invoke(status);
    }

    public void Stop()
    {
        Session? session;
        lock (_gate) session = _session;
        if (session is not null) StopSession(session, errorMessage: null);
    }

    public void Dispose() => Stop();

    /// <summary>Testhaken: ein Mixer-Takt, wie ihn der Pump-Thread ausführt.</summary>
    internal void Pump()
    {
        Session? session;
        lock (_gate) session = _session;
        if (session is not null) PumpOnce(session);
    }

    private static void DispatchToThreadPool(Action action) => Task.Run(action);

    private RecorderStatus StartLocked()
    {
        var startedAt = DateTime.Now;
        string path;
        IFlacSink sink;
        try
        {
            path = _paths.NewRecordingPath(startedAt);
            sink = _createSink(path);
        }
        catch (Exception ex) when (ex is IOException or UnauthorizedAccessException)
        {
            _log.Error("Aufnahmedatei konnte nicht angelegt werden", ex);
            return new RecorderStatus(RecorderState.Error, ErrorMessage: WriteError);
        }

        var session = new Session(path, sink, new AudioMixer(_clock, sink.Write),
            new MicSilenceDetector(_clock, MicSilenceDetector.DefaultThreshold));

        string? micId = null;
        IReadOnlyList<string> renderIds = [];
        try
        {
            micId = _backend.GetDefaultCaptureDeviceId();
            renderIds = _backend.GetActiveRenderDeviceIds();
        }
        catch (Exception ex)
        {
            _log.Error("Audiogeräte konnten nicht aufgelistet werden", ex);
        }

        if (micId is not null) TryAddSource(session, micId, SourceKind.Microphone);
        foreach (var id in renderIds) TryAddSource(session, id, SourceKind.Loopback);

        if (session.Sources.Count == 0)
        {
            sink.Dispose();
            TryDelete(path);
            _log.Error(NoDeviceError);
            return new RecorderStatus(RecorderState.Error, ErrorMessage: NoDeviceError);
        }

        session.Mixer.Start();
        session.MicDetector.Reset();
        _session = session;
        if (_runPumpThread)
        {
            session.PumpThread = new Thread(() => PumpLoop(session)) { IsBackground = true, Name = "Oreja-Mixer" };
            session.PumpThread.Start();
        }

        _log.Info($"Aufnahme gestartet: {path} ({session.Sources.Count} Quellen)");
        return new RecorderStatus(RecorderState.Recording, ComputeDegradation(session), startedAt);
    }

    private void StopSession(Session session, string? errorMessage)
    {
        lock (_gate)
        {
            if (!ReferenceEquals(_session, session)) return; // bereits gestoppt
            _session = null; // ab jetzt ignorieren Geräte-Events und Pump-Takte diese Sitzung
        }

        session.StopPump.Set();
        if (session.PumpThread is { } pump && pump != Thread.CurrentThread) pump.Join();

        // Quellen beenden, ihre FIFOs aber im Mixer lassen, damit Flush den Rest noch schreibt.
        foreach (var source in session.Sources.Keys) DisposeQuietly(source);
        session.Sources.Clear();

        try
        {
            session.Mixer.Flush();
            session.Sink.Complete();
        }
        catch (Exception ex)
        {
            _log.Error("Aufnahme konnte nicht sauber abgeschlossen werden", ex);
            errorMessage ??= ErrorMessageFor(ex);
        }
        finally
        {
            session.Sink.Dispose();
            session.StopPump.Dispose();
        }

        var status = errorMessage is null ? RecorderStatus.Idle : new RecorderStatus(RecorderState.Error, ErrorMessage: errorMessage);
        lock (_gate)
        {
            if (_session is null) _status = status; // nur, wenn inzwischen keine neue Aufnahme läuft
            else status = _status;
        }

        _log.Info($"Aufnahme beendet: {session.Path}, Pufferüberläufe: {session.Mixer.OverflowCount}" +
                  (errorMessage is null ? "" : $" ({errorMessage})"));
        StatusChanged?.Invoke(status);
        RecordingFinished?.Invoke(session.Path);
    }

    private void PumpLoop(Session session)
    {
        while (!session.StopPump.Wait(PumpInterval))
        {
            if (!PumpOnce(session)) return;
        }
    }

    private bool PumpOnce(Session session)
    {
        try
        {
            session.Mixer.Tick();
        }
        catch (Exception ex)
        {
            _log.Error("Fehler in der Aufnahme-Pipeline", ex);
            string message = ErrorMessageFor(ex);
            _dispatch(() => StopSession(session, message));
            return false;
        }

        RecorderStatus? changed;
        lock (_gate) changed = ReferenceEquals(_session, session) ? UpdateDegradationLocked(session) : null;
        if (changed is not null) StatusChanged?.Invoke(changed);
        return true;
    }

    private void OnRenderDeviceActivated(string id) => Mutate(session =>
    {
        if (!session.Sources.Keys.Any(s => s.Kind == SourceKind.Loopback && s.DeviceId == id))
            TryAddSource(session, id, SourceKind.Loopback);
    });

    private void OnDeviceDeactivated(string id) => Mutate(session =>
    {
        foreach (var source in session.Sources.Keys.Where(s => s.DeviceId == id).ToList())
            RemoveSource(session, source);
    });

    private void OnDefaultCaptureDeviceChanged(string? id) => Mutate(session =>
    {
        var mic = session.Sources.Keys.FirstOrDefault(s => s.Kind == SourceKind.Microphone);
        if (mic is not null && mic.DeviceId == id) return;
        if (mic is not null) RemoveSource(session, mic);
        if (id is not null) TryAddSource(session, id, SourceKind.Microphone);
    });

    private void OnSourceStopped(IAudioSource source, Exception? error) => Mutate(session =>
    {
        if (!session.Sources.ContainsKey(source)) return;
        _log.Error($"Quelle ausgefallen: {source.Kind} {source.DeviceId}", error);
        RemoveSource(session, source);
    });

    /// <summary>Wendet eine Änderung auf die laufende Sitzung an und meldet den neuen Zustand.</summary>
    private void Mutate(Action<Session> change)
    {
        RecorderStatus? changed;
        lock (_gate)
        {
            if (_session is null) return;
            try
            {
                change(_session);
            }
            catch (Exception ex)
            {
                _log.Error("Gerätewechsel fehlgeschlagen", ex);
            }
            changed = UpdateDegradationLocked(_session);
        }
        if (changed is not null) StatusChanged?.Invoke(changed);
    }

    private bool TryAddSource(Session session, string deviceId, SourceKind kind)
    {
        var fifo = session.Mixer.AddSource();
        IAudioSource? source = null;
        try
        {
            var observer = kind == SourceKind.Microphone ? new SampleObserver(session.MicDetector.Observe) : null;
            source = _backend.CreateSource(deviceId, kind, fifo, observer);
            source.Stopped += (s, error) => _dispatch(() => OnSourceStopped(s, error));
            source.Start();
            session.Sources.Add(source, fifo);
            if (kind == SourceKind.Microphone) session.MicDetector.Reset();
            _log.Info($"Quelle hinzugefügt: {kind} {deviceId}");
            return true;
        }
        catch (Exception ex)
        {
            session.Mixer.RemoveSource(fifo);
            if (source is not null) DisposeQuietly(source);
            _log.Error($"Quelle konnte nicht geöffnet werden: {kind} {deviceId}", ex);
            return false;
        }
    }

    private void RemoveSource(Session session, IAudioSource source)
    {
        if (!session.Sources.Remove(source, out var fifo)) return;
        session.Mixer.RemoveSource(fifo);
        DisposeQuietly(source);
        _log.Info($"Quelle entfernt: {source.Kind} {source.DeviceId}");
    }

    /// <summary>Berechnet die Einschränkung neu; liefert den neuen Status, falls er sich geändert hat.</summary>
    private RecorderStatus? UpdateDegradationLocked(Session session)
    {
        if (_status.State != RecorderState.Recording) return null;
        var degradation = ComputeDegradation(session);
        if (degradation == _status.Degradation) return null;
        _status = _status with { Degradation = degradation };
        _log.Info($"Einschränkung: {degradation}");
        return _status;
    }

    private static Degradation ComputeDegradation(Session session)
    {
        var degradation = Degradation.None;
        if (!session.Sources.Keys.Any(s => s.Kind == SourceKind.Microphone)) degradation |= Degradation.NoMicrophone;
        else if (session.MicDetector.IsSilent) degradation |= Degradation.MicrophoneSilent;
        if (!session.Sources.Keys.Any(s => s.Kind == SourceKind.Loopback)) degradation |= Degradation.NoSystemAudio;
        return degradation;
    }

    private static string ErrorMessageFor(Exception ex) =>
        ex is IOException or UnauthorizedAccessException ? WriteError : InternalError;

    private void DisposeQuietly(IAudioSource source)
    {
        try
        {
            source.Dispose();
        }
        catch (Exception ex)
        {
            _log.Error($"Quelle konnte nicht beendet werden: {source.Kind} {source.DeviceId}", ex);
        }
    }

    private void TryDelete(string path)
    {
        try
        {
            File.Delete(path);
        }
        catch (Exception ex) when (ex is IOException or UnauthorizedAccessException)
        {
            _log.Error($"{path} konnte nicht gelöscht werden", ex);
        }
    }

    private sealed class Session(string path, IFlacSink sink, AudioMixer mixer, MicSilenceDetector micDetector)
    {
        public string Path { get; } = path;
        public IFlacSink Sink { get; } = sink;
        public AudioMixer Mixer { get; } = mixer;
        public MicSilenceDetector MicDetector { get; } = micDetector;
        public Dictionary<IAudioSource, SampleFifo> Sources { get; } = [];
        public ManualResetEventSlim StopPump { get; } = new();
        public Thread? PumpThread { get; set; }
    }
}
```

```csharp
namespace Oreja.Recording;

/// <summary>Nimmt einen WASAPI-Endpunkt auf (Mikrofon oder Loopback eines Wiedergabegeräts) und schreibt in den FIFO.</summary>
public sealed class WasapiSource : IAudioSource
{
    private readonly MMDevice _device;
    private readonly WasapiCapture _capture;
    private readonly SampleConverter _converter;
    private readonly SampleFifo _fifo;
    private readonly SampleObserver? _observer;
    private volatile bool _disposed;

    /// <summary>Übernimmt den Besitz von <paramref name="device"/>.</summary>
    public WasapiSource(MMDevice device, SourceKind kind, SampleFifo fifo, SampleObserver? observer)
    {
        _device = device;
        DeviceId = device.ID;
        Kind = kind;
        _fifo = fifo;
        _observer = observer;
        _capture = kind == SourceKind.Loopback ? new WasapiLoopbackCapture(device) : new WasapiCapture(device);
        try
        {
            _converter = new SampleConverter(_capture.WaveFormat);
        }
        catch
        {
            _capture.Dispose();
            throw;
        }
        _capture.DataAvailable += OnDataAvailable;
        _capture.RecordingStopped += OnRecordingStopped;
    }

    public string DeviceId { get; }
    public SourceKind Kind { get; }

    public event Action<IAudioSource, Exception?>? Stopped;

    public void Start() => _capture.StartRecording();

    private void OnDataAvailable(object? sender, WaveInEventArgs e)
    {
        var samples = _converter.Convert(e.BufferSpan[..e.BytesRecorded]);
        _observer?.Invoke(samples);
        _fifo.Write(samples);
    }

    private void OnRecordingStopped(object? sender, StoppedEventArgs e)
    {
        if (!_disposed) Stopped?.Invoke(this, e.Exception);
    }

    public void Dispose()
    {
        _disposed = true;
        _capture.Dispose();
        _device.Dispose();
    }
}
```

```csharp
using NAudio.CoreAudioApi;
using Oreja.Infrastructure;

namespace Oreja.Recording;

/// <summary>
/// WASAPI-Zugriff: listet Endpunkte, erzeugt Quellen und übersetzt Windows-Gerätebenachrichtigungen
/// in Recorder-Events. Nur auf MTA-Threads (Thread-Pool) erzeugen und benutzen.
/// </summary>
public sealed class WasapiBackend : IAudioBackend
{
    private readonly MMDeviceEnumerator _enumerator = new();
    private readonly MMDeviceNotificationClient _notifications;
    private readonly Log _log;

    public WasapiBackend(Log log)
    {
        _log = log;
        _notifications = _enumerator.CreateNotificationClient(useSynchronizationContext: false);
        _notifications.DeviceStateChanged += OnDeviceStateChanged;
        _notifications.DeviceRemoved += (_, e) => DeviceDeactivated?.Invoke(e.DeviceId);
        _notifications.DefaultDeviceChanged += OnDefaultDeviceChanged;
    }

    public event Action<string>? RenderDeviceActivated;
    public event Action<string>? DeviceDeactivated;
    public event Action<string?>? DefaultCaptureDeviceChanged;

    public IReadOnlyList<string> GetActiveRenderDeviceIds()
    {
        var ids = new List<string>();
        foreach (var device in _enumerator.EnumerateAudioEndPoints(DataFlow.Render, DeviceState.Active))
        {
            using (device) ids.Add(device.ID);
        }
        return ids;
    }

    public string? GetDefaultCaptureDeviceId()
    {
        if (!_enumerator.TryGetDefaultAudioEndpoint(DataFlow.Capture, Role.Console, out var device) || device is null) return null;
        using (device) return device.ID;
    }

    public IAudioSource CreateSource(string deviceId, SourceKind kind, SampleFifo fifo, SampleObserver? observer)
    {
        var device = _enumerator.GetDevice(deviceId);
        try
        {
            return new WasapiSource(device, kind, fifo, observer);
        }
        catch
        {
            device.Dispose();
            throw;
        }
    }

    private void OnDeviceStateChanged(object? sender, DeviceStateChangedEventArgs e)
    {
        if (e.NewState != DeviceState.Active)
        {
            DeviceDeactivated?.Invoke(e.DeviceId);
            return;
        }

        // Im Benachrichtigungs-Callback keine Enumerator-Aufrufe: Abfrage auf den Thread-Pool verlagern.
        string id = e.DeviceId;
        Task.Run(() =>
        {
            try
            {
                using var device = _enumerator.GetDevice(id);
                if (device.DataFlow == DataFlow.Render) RenderDeviceActivated?.Invoke(id);
            }
            catch (Exception ex)
            {
                _log.Error($"Gerät {id} konnte nicht abgefragt werden", ex);
            }
        });
    }

    private void OnDefaultDeviceChanged(object? sender, DefaultDeviceChangedEventArgs e)
    {
        if (e.Flow == DataFlow.Capture && e.Role == Role.Console)
            DefaultCaptureDeviceChanged?.Invoke(string.IsNullOrEmpty(e.DeviceId) ? null : e.DeviceId);
    }

    public void Dispose()
    {
        _notifications.Dispose();
        _enumerator.Dispose();
    }
}
```

```csharp
namespace Oreja;

public enum TrayIconKind { Idle, Recording, RecordingDegraded, Error }

/// <summary>Bildet den Recorder-Status auf Icon, Tooltip und Menütext ab.</summary>
public static class StatusText
{
    private const int MaxTooltipLength = 127;

    public static TrayIconKind IconKind(RecorderStatus status) => status.State switch
    {
        RecorderState.Recording when status.Degradation != Degradation.None => TrayIconKind.RecordingDegraded,
        RecorderState.Recording => TrayIconKind.Recording,
        RecorderState.Error => TrayIconKind.Error,
        _ => TrayIconKind.Idle,
    };

    public static string Tooltip(RecorderStatus status)
    {
        string text = status.State switch
        {
            RecorderState.Recording =>
                $"Oreja – Aufnahme seit {status.StartedAt?.ToString("HH:mm", CultureInfo.InvariantCulture)}{DegradationSuffix(status.Degradation)}",
            RecorderState.Error => $"Oreja – Fehler: {status.ErrorMessage}",
            _ => "Oreja – bereit",
        };
        return text.Length <= MaxTooltipLength ? text : text[..MaxTooltipLength];
    }

    public static string ToggleMenuText(RecorderStatus status) =>
        status.State == RecorderState.Recording ? "Aufnahme stoppen" : "Aufnahme starten";

    private static string DegradationSuffix(Degradation degradation)
    {
        var parts = new List<string>();
        if (degradation.HasFlag(Degradation.NoMicrophone)) parts.Add("ohne Mikrofon");
        if (degradation.HasFlag(Degradation.MicrophoneSilent)) parts.Add("Mikrofon stumm/gesperrt");
        if (degradation.HasFlag(Degradation.NoSystemAudio)) parts.Add("ohne Systemaudio");
        return parts.Count == 0 ? "" : $" ({string.Join(", ", parts)})";
    }
}
```

```csharp
    private readonly long _intervalMilliseconds;

    public ClickGuard(TimeSpan interval) => _intervalMilliseconds = (long)interval.TotalMilliseconds;

    public bool TryAccept(long nowMilliseconds)

        if (_lastAccepted is long last && nowMilliseconds - last < _intervalMilliseconds) return false;
```

```csharp
using System.Drawing.Drawing2D;
using System.Runtime.InteropServices;

namespace Oreja;

/// <summary>Zeichnet die vier Tray-Icons in der gewünschten Größe (keine Bilddateien nötig).</summary>
public sealed class TrayIcons : IDisposable
{
    private static readonly Color Grey = Color.FromArgb(0x8A, 0x8A, 0x8A);
    private static readonly Color Red = Color.FromArgb(0xE5, 0x39, 0x35);
    private static readonly Color Yellow = Color.FromArgb(0xFF, 0xC1, 0x07);
    private static readonly Color Outline = Color.FromArgb(0x20, 0x20, 0x20);

    private readonly Dictionary<TrayIconKind, Icon> _icons = [];
    private readonly List<IntPtr> _handles = [];

    public TrayIcons(int size)
    {
        foreach (var kind in Enum.GetValues<TrayIconKind>()) _icons[kind] = Draw(kind, size);
    }

    public Icon this[TrayIconKind kind] => _icons[kind];

    private Icon Draw(TrayIconKind kind, int s)
    {
        using var bitmap = new Bitmap(s, s);
        using (var g = Graphics.FromImage(bitmap))
        {
            g.SmoothingMode = SmoothingMode.AntiAlias;
            g.Clear(Color.Transparent);
            float pad = s / 8f;
            var circle = new RectangleF(pad, pad, s - 2 * pad, s - 2 * pad);

            if (kind is TrayIconKind.Idle or TrayIconKind.Error)
            {
                using var ring = new Pen(Grey, Math.Max(2f, s / 8f));
                g.DrawEllipse(ring, circle);
            }
            else
            {
                using var fill = new SolidBrush(Red);
                g.FillEllipse(fill, circle);
            }

            using var yellow = new SolidBrush(Yellow);
            using var outline = new Pen(Outline, 1f);
            if (kind == TrayIconKind.RecordingDegraded)
            {
                float d = s * 0.45f;
                var dot = new RectangleF(s - d, s - d, d - 1, d - 1);
                g.FillEllipse(yellow, dot);
                g.DrawEllipse(outline, dot);
            }
            else if (kind == TrayIconKind.Error)
            {
                float h = s * 0.55f;
                PointF[] triangle = [new(s - h / 2 - 0.5f, s - h), new(s - 0.5f, s - 0.5f), new(s - h - 0.5f, s - 0.5f)];
                g.FillPolygon(yellow, triangle);
                g.DrawPolygon(outline, triangle);
            }
        }

        IntPtr handle = bitmap.GetHicon();
        _handles.Add(handle);
        return Icon.FromHandle(handle);
    }

    public void Dispose()
    {
        foreach (var icon in _icons.Values) icon.Dispose();
        foreach (var handle in _handles) DestroyIcon(handle);
        _icons.Clear();
        _handles.Clear();
    }

    [DllImport("user32.dll")]
    private static extern bool DestroyIcon(IntPtr handle);
}
```

```csharp
using System.Diagnostics;
using Microsoft.Win32;
using Oreja.Infrastructure;
using Oreja.Recording;

namespace Oreja;

/// <summary>Das Tray-Icon: Linksklick schaltet die Aufnahme, Rechtsklick öffnet das Menü. Keine Fenster, keine Töne.</summary>
public sealed class TrayApp : ApplicationContext
{
    private static readonly TimeSpan StopTimeout = TimeSpan.FromSeconds(2);
    private static readonly TimeSpan MoveTimeoutOnExit = TimeSpan.FromSeconds(10);

    private readonly Recorder _recorder;
    private readonly Settings _settings;
    private readonly Autostart _autostart;
    private readonly OutputPaths _paths;
    private readonly RecordingMover _mover;
    private readonly Log _log;
    private readonly SynchronizationContext _ui;
    private readonly TrayIcons _icons;
    private readonly NotifyIcon _notifyIcon;
    private readonly ToolStripMenuItem _toggleItem;
    private readonly ToolStripMenuItem _autostartItem;
    private readonly ClickGuard _clickGuard = new(TimeSpan.FromSeconds(1));
    private readonly object _movesGate = new();
    private Task _moves = Task.CompletedTask;
    private bool _busy;
    private bool _exiting;

    public TrayApp(Recorder recorder, Settings settings, Autostart autostart, OutputPaths paths, RecordingMover mover, Log log)
    {
        _recorder = recorder;
        _settings = settings;
        _autostart = autostart;
        _paths = paths;
        _mover = mover;
        _log = log;
        _ui = SynchronizationContext.Current ?? throw new InvalidOperationException("UI-SynchronizationContext fehlt");
        _icons = new TrayIcons(SystemInformation.SmallIconSize.Width);

        string version = typeof(TrayApp).Assembly.GetName().Version!.ToString(3);
        _toggleItem = new ToolStripMenuItem("Aufnahme starten", null, (_, _) => Toggle());
        _autostartItem = new ToolStripMenuItem("Mit Windows starten", null, (_, _) => ToggleAutostart())
        {
            Checked = _autostart.IsEnabled,
        };
        var menu = new ContextMenuStrip();
        menu.Items.AddRange(
        [
            new ToolStripMenuItem($"Oreja {version}") { Enabled = false },
            new ToolStripSeparator(),
            _toggleItem,
            new ToolStripMenuItem("Aufnahmeordner öffnen", null, (_, _) => OpenOutputFolder()),
            new ToolStripMenuItem("Aufnahmeordner ändern…", null, (_, _) => ChangeOutputFolder()),
            _autostartItem,
            new ToolStripSeparator(),
            new ToolStripMenuItem("Beenden", null, (_, _) => Exit()),
        ]);

        _notifyIcon = new NotifyIcon { ContextMenuStrip = menu };
        _notifyIcon.MouseClick += (_, e) =>
        {
            if (e.Button == MouseButtons.Left) Toggle();
        };

        _recorder.StatusChanged += status => _ui.Post(_ => Render(status), null);
        _recorder.RecordingFinished += _ => QueueMove();
        SystemEvents.PowerModeChanged += OnPowerModeChanged;
        SystemEvents.SessionEnding += OnSessionEnding;

        Render(_recorder.Status);
        _notifyIcon.Visible = true;
        QueueMove(); // Reste nach Absturz oder nicht erreichbarem Zielordner
    }

    private void Render(RecorderStatus status)
    {
        if (_exiting) return;
        _notifyIcon.Icon = _icons[StatusText.IconKind(status)];
        _notifyIcon.Text = StatusText.Tooltip(status);
        _toggleItem.Text = StatusText.ToggleMenuText(status);
    }

    private async void Toggle()
    {
        if (_busy || _exiting || !_clickGuard.TryAccept(Environment.TickCount64)) return;
        _busy = true;
        try
        {
            if (_recorder.Status.State == RecorderState.Recording) await Task.Run(_recorder.Stop);
            else await Task.Run(_recorder.Start);
        }
        catch (Exception ex)
        {
            _log.Error("Umschalten der Aufnahme fehlgeschlagen", ex);
            Render(new RecorderStatus(RecorderState.Error, ErrorMessage: "Interner Fehler – siehe Protokoll"));
        }
        finally
        {
            _busy = false;
        }
    }

    private void QueueMove()
    {
        lock (_movesGate)
        {
            _moves = _moves.ContinueWith(_ =>
            {
                try
                {
                    _mover.MovePending();
                }
                catch (Exception ex)
                {
                    _log.Error("Verschieben fehlgeschlagen", ex);
                }
            }, TaskScheduler.Default);
        }
    }

    private void OpenOutputFolder()
    {
        try
        {
            string folder = _paths.TargetFolder;
            Directory.CreateDirectory(folder);
            Process.Start(new ProcessStartInfo("explorer.exe", $"\"{folder}\"") { UseShellExecute = true });
        }
        catch (Exception ex)
        {
            _log.Error("Aufnahmeordner konnte nicht geöffnet werden", ex);
        }
    }

    private void ChangeOutputFolder()
    {
        using var dialog = new FolderBrowserDialog
        {
            Description = "Aufnahmeordner wählen",
            UseDescriptionForTitle = true,
            SelectedPath = _settings.OutputFolder,
            ShowNewFolderButton = true,
        };
        if (dialog.ShowDialog() != DialogResult.OK) return;

        _settings.OutputFolder = dialog.SelectedPath;
        _log.Info($"Aufnahmeordner geändert: {dialog.SelectedPath}");
        QueueMove();
    }

    private void ToggleAutostart()
    {
        try
        {
            if (_autostart.IsEnabled) _autostart.Disable();
            else _autostart.Enable();
        }
        catch (Exception ex)
        {
            _log.Error("Autostart konnte nicht geändert werden", ex);
        }
        _autostartItem.Checked = _autostart.IsEnabled;
    }

    private void OnPowerModeChanged(object? sender, PowerModeChangedEventArgs e)
    {
        if (e.Mode == PowerModes.Suspend) StopAndWait("Standby");
    }

    private void OnSessionEnding(object? sender, SessionEndingEventArgs e) => StopAndWait("Abmelden/Herunterfahren");

    private void StopAndWait(string reason)
    {
        if (_recorder.Status.State != RecorderState.Recording) return;
        _log.Info($"Aufnahme wird gestoppt: {reason}");
        if (!Task.Run(_recorder.Stop).Wait(StopTimeout))
            _log.Error($"Stoppen dauerte länger als {StopTimeout.TotalSeconds} s");
    }

    private void Exit()
    {
        if (_exiting) return;
        StopAndWait("Beenden");
        _exiting = true;

        Task moves;
        lock (_movesGate) moves = _moves;
        if (!moves.Wait(MoveTimeoutOnExit))
            _log.Info("Verschieben beim Beenden abgebrochen, wird beim nächsten Start fortgesetzt");

        ExitThread();
    }

    protected override void Dispose(bool disposing)
    {
        if (disposing)
        {
            SystemEvents.PowerModeChanged -= OnPowerModeChanged;
            SystemEvents.SessionEnding -= OnSessionEnding;
            _notifyIcon.Visible = false;
            _notifyIcon.ContextMenuStrip?.Dispose();
            _notifyIcon.Dispose();
            _icons.Dispose();
        }
        base.Dispose(disposing);
    }
}
```

```csharp
using Microsoft.Win32;
using Oreja.Infrastructure;
using Oreja.Recording;

namespace Oreja;

internal static class Program
{
    [STAThread]
    private static void Main()
    {
        using var singleInstance = new Mutex(initiallyOwned: true, @"Local\Oreja.SingleInstance", out bool isFirstInstance);
        if (!isFirstInstance) return;

        ApplicationConfiguration.Initialize();
        SynchronizationContext.SetSynchronizationContext(new WindowsFormsSynchronizationContext());

        string appData = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "Oreja");
        var log = new Log(Path.Combine(appData, "oreja.log"));
        Application.ThreadException += (_, e) => log.Error("Unbehandelter Fehler (UI)", e.Exception);
        AppDomain.CurrentDomain.UnhandledException += (_, e) => log.Error("Unbehandelter Fehler", e.ExceptionObject as Exception);
        log.Info($"Oreja {typeof(Program).Assembly.GetName().Version!.ToString(3)} gestartet");

        using var settingsKey = Registry.CurrentUser.CreateSubKey(@"Software\Oreja");
        using var runKey = Registry.CurrentUser.CreateSubKey(Autostart.RunKeyPath);
        var settings = new Settings(settingsKey);
        var autostart = new Autostart(runKey, "Oreja", Environment.ProcessPath!);
        if (!settings.IsInitialized)
        {
            autostart.Enable();
            settings.MarkInitialized();
        }
        else
        {
            autostart.RepairPath();
        }

        var paths = new OutputPaths(Path.Combine(appData, "recording"), () => settings.OutputFolder);
        var mover = new RecordingMover(paths, log);

        // WASAPI-Objekte müssen im MTA leben, deshalb entsteht das Backend auf einem Thread-Pool-Thread.
        var backend = Task.Run(() => new WasapiBackend(log)).GetAwaiter().GetResult();
        var recorder = new Recorder(backend, new SystemClock(), path => LibFlacSink.Create(path), paths, log);

        using (var app = new TrayApp(recorder, settings, autostart, paths, mover, log))
        {
            Application.Run(app);
        }

        Task.Run(() =>
        {
            recorder.Dispose();
            backend.Dispose();
        }).Wait(TimeSpan.FromSeconds(3));
        log.Info("Oreja beendet");
    }
}
```
